  function aName(g) {

  }
  function atlasWalk() {

  // the action at a shard: what the run it calls or spawns does, read from that run's own SQL. Each BUILD span is one paste to mother at the SQL prompt;
  // each statement in it is named by what it does: a cell written (cell ← where from), a table read, a column rewritten, a row laid, a run run, a source asked, a line typed
  function topSplit(s, sep) {
    var out = [], d = 0, qt = false, cur = '';
    for (var i = 0; i < s.length; i++) { var c = s.charAt(i);
      if (c === "'") qt = !qt;
      if (!qt) { if (c === '(') d++; else if (c === ')') d--; else if (c === sep && d === 0) { out.push(cur.trim()); cur = ''; continue; } }
      cur += c; }
    if (cur.trim()) out.push(cur.trim()); return out;
  }
  function cellWord(x) { var m; x = x.trim(); if (/^\d+$/.test(x)) return aName(+x); if ((m = /symbol = '([^']+)'/i.exec(x))) return m[1]; return x.replace(/^\(\s*|\s*\)$/g, '').slice(0, 24); }
  function fromWhere(v, state) {
    if (/^NULL\b/i.test(v)) return state === '0' ? null : 'nothing';
    if (/^reply\b/i.test(v)) return "mother's answer";
    if (/FROM seq\b.*\brun = above\b/i.test(v)) return 'the line';
    if (/^(what|newgate|to|from|me|above)\b/i.test(v)) return v.split(/\s/)[0];
    if (/\bcount \(/i.test(v)) return 'a count';
    if (/run_memory/i.test(v)) return 'a memory row';
    if (/\bdna\b/i.test(v)) return 'a dna read';
    if (/^\d+$/.test(v)) return v;
    return 'a read';
  }
  function stmtAct(st) {
    var s = st.replace(/\s+/g, ' ').trim(), m;
    if (!s || /^SELECT changes \(/i.test(s)) return null;
    if ((m = /^INSERT OR REPLACE INTO (?:main\.|sixth\.)?run_sheet \( ?run , cell , state , value ?\) (VALUES|SELECT) (.*)$/i.exec(s))) {
      var body = m[2]; if (m[1].toUpperCase() === 'VALUES') body = body.replace(/^\(\s*/, '').replace(/\s*\)\s*$/, '');
      var it = topSplit(body, ','); if (it.length < 4) return 'write a cell';
      var src = fromWhere(it.slice(3).join(' , '), it[2]); if (src === null) return null;
      var on = it[0] === 'above' ? '' : it[0] === 'me' ? ' (its own sheet)' : ' (on run ' + it[0] + ')';
      return cellWord(it[1]) + ' ← ' + src + on;
    }
    if ((m = /^INSERT INTO (?:main\.)?replay \( ?parent , source , kind ?\) (?:VALUES \( ?|SELECT )(\w+) , (\d+) , (\d)/i.exec(s))) return 'run r' + m[2] + (m[3] === '2' ? ' as a spawn' : '') + (/WHERE NOT EXISTS/i.test(s) ? ' if not born' : '');
    if ((m = /^INSERT INTO (\w+)\.inbox\b/i.exec(s))) return 'ask ' + m[1];
    if (/^INSERT INTO (?:main\.)?third\b/i.test(s)) return 'type lines into the main brain';
    if (/^INSERT INTO (?:main\.)?input\b/i.test(s)) return 'type a line';
    if ((m = /^INSERT (?:OR \w+ )?INTO ([\w.]+)/i.exec(s))) return 'lay a ' + m[1].replace(/^main\./, '') + ' row';
    if ((m = /^UPDATE ([\w.]+) SET (\w+)/i.exec(s))) return 'rewrite ' + m[1].replace(/^main\./, '') + '.' + m[2];
    if ((m = /^DELETE FROM ([\w.]+)/i.exec(s))) return 'delete ' + m[1].replace(/^main\./, '') + ' rows';
    if (/^SELECT mother \(/i.test(s)) return /FROM "/.test(s) ? 'read every row of the table the subject names (mother runs it)' : 'mother runs the text it builds';
    if (/^(SELECT|WITH)\b/i.test(s)) {
      var t = {}, re = /\b(?:FROM|JOIN) ([a-z_][\w.]*)/ig, x; while ((x = re.exec(s))) { var nm = x[1].replace(/^main\./, ''); if (!/^(select|seq|label|back|g|x|z|c|q|p|kb|br|mk|nm|pr|tok|gs|sp|oc|ok|sr|d|s|r|h|n|l|j|json_each)$/i.test(nm)) t[nm] = 1; }
      var tl = Object.keys(t); if (!tl.length) tl = /\bseq\b/i.test(s) ? ['the line'] : ['a value'];
      var key = /WHERE (?:\w+\.)?gate = what\b/i.test(s) ? ' of the subject' : /WHERE (?:\w+\.)?run = what\b/i.test(s) ? ' of run what' : /^WITH RECURSIVE back\b/i.test(s) ? ' back up the parents of run what' : '';
      return 'read ' + tl.join(', ') + key;
    }
    if ((m = /^CREATE (\w+)/i.exec(s))) return 'create a ' + m[1].toLowerCase();
    return s.split(' ').slice(0, 4).join(' ');
  }
  function actOf(run) {
    if (ATACT[run] !== undefined) return ATACT[run];
    var t = TEXT[run]; if (t === undefined) return null;
    var parts = t.split(/\bBUILD\b/), spans = [];
    for (var i = 1; i < parts.length; i += 2) {
      var acts = [];
      topSplit(parts[i], ';').forEach(function (st) { var a = stmtAct(st); if (!a) return; var last = acts[acts.length - 1]; if (last && /^ask /.test(a) && /^ask /.test(last)) acts[acts.length - 1] = last + ', ' + a.slice(4); else if (last !== a) acts.push(a); });
      if (acts.length) spans.push(acts);
    }
    return ATACT[run] = { spans: spans, text: t };
  }
  function actLine(run, full) {
    var a = actOf(run); if (!a) return 'r' + run + ' (not read)';
    var s = a.spans.length ? a.spans.map(function (x) { return x.join(' · '); }).join(' ▸ ') : 'its line, no BUILD span: ' + a.text.slice(0, 90);
    return full || s.length <= 220 ? s : s.slice(0, 220) + '…';
  }

  function actCell(g) {

  function atlasView() {

      else if (show === 'drawn') { var nd = NODE[String(g)]; if (!(nd && nd.mesh && nd.mesh.visible)) continue; }
      else if (show === 'loaded') { if (LAB[g] === undefined) continue; }
      out.push(i);
    }
    ATVIEW = out;
  }
