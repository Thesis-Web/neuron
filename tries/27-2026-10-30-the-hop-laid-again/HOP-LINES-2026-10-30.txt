BUILD DROP TRIGGER land_go ; CREATE TRIGGER land_go AFTER INSERT ON land BEGIN INSERT OR REPLACE INTO step (k, v) VALUES (1, mother('UPDATE ' || 'main."' || NEW.gate || '"' || ' SET times = times + 1 WHERE before IS ? AND trigger IS ? RETURNING shard, call, spawn', (CASE WHEN NEW.pos = 0 OR NEW.via = 1 OR NEW.gate = 3 THEN NULL ELSE coalesce(NEW.kb, (SELECT gate FROM seq WHERE run = NEW.run AND pos < NEW.pos AND gate <> 3 ORDER BY pos DESC LIMIT 1), (SELECT arrival FROM run WHERE id = NEW.run)) END), (CASE WHEN NEW.pos = 0 OR NEW.via = 1 OR NEW.gate = 3 THEN NULL WHEN NEW.kn IS NULL THEN (SELECT gate FROM seq WHERE run = NEW.run AND pos > NEW.pos AND gate <> 3 ORDER BY pos LIMIT 1) WHEN NEW.kn = 0 THEN NULL ELSE NEW.kn END))); INSERT OR REPLACE INTO step (k, v) SELECT 1, mother('INSERT INTO ' || 'main."' || NEW.gate || '"' || ' (shard, before, trigger, times) SELECT n + 1, ?, ?, 1 FROM main.shards RETURNING shard, call, spawn', (CASE WHEN NEW.pos = 0 OR NEW.via = 1 OR NEW.gate = 3 THEN NULL ELSE coalesce(NEW.kb, (SELECT gate FROM seq WHERE run = NEW.run AND pos < NEW.pos AND gate <> 3 ORDER BY pos DESC LIMIT 1), (SELECT arrival FROM run WHERE id = NEW.run)) END), (CASE WHEN NEW.pos = 0 OR NEW.via = 1 OR NEW.gate = 3 THEN NULL WHEN NEW.kn IS NULL THEN (SELECT gate FROM seq WHERE run = NEW.run AND pos > NEW.pos AND gate <> 3 ORDER BY pos LIMIT 1) WHEN NEW.kn = 0 THEN NULL ELSE NEW.kn END)) WHERE (SELECT v FROM step WHERE k = 1) = 'ok'; UPDATE shards SET n = n + 1 WHERE (SELECT v FROM step WHERE k = 1) <> 'ok' AND CAST(substr((SELECT v FROM step WHERE k = 1), 1, instr((SELECT v FROM step WHERE k = 1), ' | ') - 1) AS INTEGER) = n + 1; INSERT INTO hit (run, pos, gate, before, next, via, had, shard, call, spawn, kb, kn) SELECT NEW.run, NEW.pos, NEW.gate, NEW.before, NEW.next, NEW.via, (CASE WHEN NEW.next IS NULL THEN 0 WHEN EXISTS (SELECT 1 FROM pragma_table_info(CAST(NEW.next AS TEXT), 'main')) THEN 1 ELSE 0 END), CAST(substr(v, 1, instr(v, ' | ') - 1) AS INTEGER), nullif(substr(v, instr(v, ' | ') + 3, instr(substr(v, instr(v, ' | ') + 3), ' | ') - 1), ''), nullif(substr(substr(v, instr(v, ' | ') + 3), instr(substr(v, instr(v, ' | ') + 3), ' | ') + 3), ''), (CASE WHEN NEW.pos = 0 OR NEW.via = 1 OR NEW.gate = 3 THEN NULL ELSE coalesce(NEW.kb, (SELECT gate FROM seq WHERE run = NEW.run AND pos < NEW.pos AND gate <> 3 ORDER BY pos DESC LIMIT 1), (SELECT arrival FROM run WHERE id = NEW.run)) END), (CASE WHEN NEW.pos = 0 OR NEW.via = 1 OR NEW.gate = 3 THEN NULL WHEN NEW.kn IS NULL THEN coalesce((SELECT gate FROM seq WHERE run = NEW.run AND pos > NEW.pos AND gate <> 3 ORDER BY pos LIMIT 1), 0) WHEN NEW.kn = 0 THEN NULL ELSE NEW.kn END) FROM step WHERE k = 1 AND v <> 'ok' AND v NOT LIKE 'error:%'; END ; BUILD
BUILD DROP TRIGGER hit_go ; CREATE TRIGGER hit_go AFTER INSERT ON hit BEGIN INSERT INTO run_receipt (run, pos, gate, shard) VALUES (NEW.run, NEW.pos, NEW.gate, NEW.shard); INSERT INTO hop (text, p1, p2, p3) SELECT 'INSERT INTO main.ended (run, gate, shard) VALUES (?, ?, ?)', NEW.run, NEW.gate, NEW.shard WHERE NEW.next IS NULL AND NEW.via = 0;
