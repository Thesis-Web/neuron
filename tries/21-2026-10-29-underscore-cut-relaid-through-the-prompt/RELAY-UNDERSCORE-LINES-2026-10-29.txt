INSERT INTO seq (run, pos, gate) SELECT (SELECT max(id) FROM run), n, (SELECT gate FROM label WHERE symbol = s) FROM (WITH RECURSIVE ch(i, c) AS (SELECT 1, substr(NEW.text, 1, 1) WHERE NEW.text <> '' UNION ALL SELECT i + 1, substr(NEW.text, i + 1, 1) FROM ch WHERE i < length(NEW.text)), m AS (SELECT i, c, (c GLOB '[A-Za-z]') AS w FROM ch), st AS (SELECT i, c, w, CASE WHEN w = 0 OR i = 1 OR lag(w) OVER (ORDER BY i) = 0 THEN 1 ELSE 0 END AS start FROM m), sym AS (SELECT i, c, sum(start) OVER (ORDER BY i ROWS UNBOUNDED PRECEDING) AS n FROM st) SELECT n, group_concat(c, '') AS s FROM (SELECT * FROM sym ORDER BY i) GROUP BY n ORDER BY n); INSERT INTO land (run, pos, gate, before, next, via) SELECT (SELECT max(id) FROM run), 0, (SELECT arrival FROM run WHERE id = (SELECT max(id) FROM run)), NULL, (SELECT gate FROM seq WHERE run = (SELECT max(id) FROM run) AND pos = 1), 0 WHERE EXISTS (SELECT 1 FROM seq WHERE run = (SELECT max(id) FROM run) AND pos = 1); END ; BUILD
