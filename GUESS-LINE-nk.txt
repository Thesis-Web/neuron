# GUESS-LINE-nk.txt (2026-10-30, mine): the guesser picks its companion by how exclusively the two words keep company (c*c over the lines of
# each), letters only, ties by count; confidence stays the share of the asked word's lines that hold the guess, belief at most 0.5 on first
# use (the owner's words, 10-28). Before, it picked the companion that kept the most company, which named the scaffolding in every line
# (Webster, Appendix, en). Typed once at the user's prompt; the guesser's trigger is a row of the guess source, so a line lays it again.
BUILD DROP TRIGGER guess.work ; CREATE TRIGGER guess.work AFTER INSERT ON inbox BEGIN INSERT INTO guesses ( word , guess , confidence , belief , used , asked_by , cell , asked_returns ) SELECT NEW.word , g.word , g.c * 1.0 / t.n , min ( g.c * 1.0 / t.n , 0.5 ) , 1 , NEW.run , NEW.cell , NEW.returns FROM ( SELECT word , count ( * ) AS c FROM corpus WHERE line IN ( SELECT line FROM corpus WHERE word = NEW.word ) AND word <> NEW.word AND word GLOB '[A-Za-z]*' AND length ( word ) > 1 GROUP BY word ) g , ( SELECT count ( DISTINCT line ) AS n FROM corpus WHERE word = NEW.word ) t ORDER BY g.c * 1.0 * g.c / ( t.n * ( SELECT count ( * ) FROM corpus x WHERE x.word = g.word ) ) DESC , g.c DESC , g.word LIMIT 1 ; INSERT INTO guesses ( word , guess , confidence , belief , used , asked_by , cell , asked_returns ) SELECT NEW.word , NULL , 0 , 0 , 1 , NEW.run , NEW.cell , NEW.returns WHERE NOT EXISTS ( SELECT 1 FROM corpus WHERE word = NEW.word ) ; INSERT INTO outbox ( run , cell , about , text ) SELECT NEW.run , NEW.cell , NEW.about , guess || ' ' || printf ( '%.2f' , confidence ) || ' ' || printf ( '%.2f' , belief ) FROM guesses WHERE id = ( SELECT max ( id ) FROM guesses ) AND guess IS NOT NULL AND confidence > 0 ; DELETE FROM inbox WHERE id = NEW.id ; END ; BUILD
